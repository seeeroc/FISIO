; Archivo de prueba FISIO con errores de alfabeto reales
; Todos los siguientes símbolos están fuera del alfabeto permitido en FISIO.
; El analizador léxico actual reportará cada uno como un error de símbolo no reconocido.

; ERROR 1: Uso de '@'
x := 5 @ 2;

; ERROR 2: Uso de '#'
y := 3 # 14;

; ERROR 3: Uso de '$'
precio := 100$;

; ERROR 4: Uso de '%'
porcentaje := 50%;

; ERROR 5: Uso de '&'
y_comercial := a & b;

; ERROR 6: Uso de guion bajo '_' (no pertenece al alfabeto FISIO)
mi_variable := 25;

; ERROR 7: Uso de llaves '{' y '}'
conjunto := { 1, 2, 3 };

; ERROR 8: Uso de comillas '"' (no se permiten strings)
mensaje := "hola";

; ERROR 9: Uso de barra invertida '\' y pipe '|'
operador_invalido := a \ b | c;

; ERROR 10: Signos de interrogación '¿' '?' y exclamación '¡'
pregunta := ¿test?;
info := ¡alerta!;
